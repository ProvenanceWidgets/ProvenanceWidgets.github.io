# Dropdown

A Dropdown is a user interface control element that allows users to select one option from a set of options.

```jsx
import { useState } from "react";
import {
  ProvenanceButton,
  SingleSelectDropdown,
} from "provenance-widgets";

const options = [
  { label: "New York", value: "New York" },
  { label: "Rome", value: "Rome" },
  { label: "London", value: "London" },
  { label: "Istanbul", value: "Istanbul" },
  { label: "Paris", value: "Paris" },
];

export default function DropdownExample() {
  const [selected, setSelected] = useState(options[0]);

  const provenanceChange = provenance => {
    console.log(provenance);
  };

  const selectedChange = nextSelected => {
    console.log(nextSelected);
    setSelected(nextSelected);
  };

  return (
    <div>
      <ProvenanceButton target="jsx-provenance-dropdown" />
      <SingleSelectDropdown
        id="jsx-provenance-dropdown"
        name="jsx-provenance-dropdown"
        options={options}
        selected={selected}
        optionLabel="label"
        dataKey="value"
        freeze={false}
        visualize={true}
        onProvenanceChange={provenanceChange}
        onSelectedChange={selectedChange}
      />
    </div>
  );
}
```

```js
// Create a new instance of the Dropdown
const dropdown = document.createElement("web-provenance-dropdown");

const options = [
  { label: "New York", value: "New York" },
  { label: "Rome", value: "Rome" },
  { label: "London", value: "London" },
  { label: "Istanbul", value: "Istanbul" },
  { label: "Paris", value: "Paris" },
];

// Set the properties
dropdown.id = "webcomponent-provenance-dropdown";
dropdown.options = options;
dropdown.selected = options[0];
dropdown.optionLabel = "label";
dropdown.dataKey = "value";
dropdown.freeze = false;
dropdown.visualize = true;

dropdown.addEventListener("provenanceChange", event => {
  console.log(event.detail);
});

dropdown.addEventListener("selectedChange", event => {
  console.log(event.detail);
});

// Append it to the DOM
document.body.appendChild(dropdown);
```

### Events

In React, use `onSelectedChange` and `onProvenanceChange`. Web Components use `selectedChange` and `provenanceChange`, with emitted values available in `event.detail`.
