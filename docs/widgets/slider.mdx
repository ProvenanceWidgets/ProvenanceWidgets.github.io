# Slider

A Slider is a user interface control element that allows users to select a numeric value or range from a given range.

```jsx
import { useState } from "react";
import { ProvenanceButton, Rangeslider } from "provenance-widgets";

const options = {
  floor: 0,
  ceil: 250,
  showTicks: true,
  tickStep: 25,
};

export default function SliderExample() {
  const [range, setRange] = useState([50, 150]);

  const provenanceChange = provenance => {
    console.log(provenance);
  };

  const selectedChange = nextRange => {
    console.log(nextRange);
    setRange(nextRange);
  };

  return (
    <div>
      <ProvenanceButton target="jsx-provenance-slider" />
      <Rangeslider
        id="jsx-provenance-slider"
        value={range[0]}
        highValue={range[1]}
        options={options}
        freeze={false}
        visualize={true}
        onProvenanceChange={provenanceChange}
        onSelectedChange={selectedChange}
      />
    </div>
  );
}
```

```js
// Create a new instance of the Slider
const slider = document.createElement("web-provenance-slider");

// Set the properties
slider.id = "webcomponent-provenance-slider";
slider.value = 50;
slider.highValue = 150;
slider.options = {
  floor: 0,
  ceil: 250,
  showTicks: true,
  tickStep: 25,
};
slider.freeze = false;
slider.visualize = true;

slider.addEventListener("provenanceChange", event => {
  console.log(event.detail);
});

slider.addEventListener("selectedChange", event => {
  console.log(event.detail);
});

// Append it to the DOM
document.body.appendChild(slider);
```

### Events

In React, use `onSelectedChange` and `onProvenanceChange`. Web Components use `selectedChange` and `provenanceChange`, with emitted values available in `event.detail`.
