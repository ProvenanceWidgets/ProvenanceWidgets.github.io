# Radio Button

A Radio Button is a user interface control element that allows users to select one option from a set of options.

```jsx
import { useState } from "react";
import { ProvenanceButton, RadioGroup } from "provenance-widgets";

const data = [
  { label: "New York", value: "New York" },
  { label: "Rome", value: "Rome" },
  { label: "London", value: "London" },
  { label: "Istanbul", value: "Istanbul" },
  { label: "Paris", value: "Paris" },
];

export default function RadioButtonExample() {
  const [selected, setSelected] = useState(data[0].value);

  const provenanceChange = provenance => {
    console.log(provenance);
  };

  const selectedChange = nextSelected => {
    console.log(nextSelected);
    setSelected(nextSelected);
  };

  return (
    <div>
      <ProvenanceButton target="jsx-provenance-radiobutton" />
      <RadioGroup
        id="jsx-provenance-radiobutton"
        name="jsx-provenance-radiobutton"
        data={data}
        selected={selected}
        freeze={false}
        visualize={true}
        onProvenanceChange={provenanceChange}
        onSelectedChange={selectedChange}
      />
    </div>
  );
}
```

```js
// Create a new instance of the Radio Button group
const radiobutton = document.createElement(
  "web-provenance-radiobutton",
);

const data = [
  { label: "New York", value: "New York" },
  { label: "Rome", value: "Rome" },
  { label: "London", value: "London" },
  { label: "Istanbul", value: "Istanbul" },
  { label: "Paris", value: "Paris" },
];

// Set the properties
radiobutton.id = "webcomponent-provenance-radiobutton";
radiobutton.name = "webcomponent-provenance-radiobutton";
radiobutton.data = data;
radiobutton.selected = data[0].value;
radiobutton.freeze = false;
radiobutton.visualize = true;

radiobutton.addEventListener("provenanceChange", event => {
  console.log(event.detail);
});

radiobutton.addEventListener("selectedChange", event => {
  console.log(event.detail);
});

// Append it to the DOM
document.body.appendChild(radiobutton);
```

### Events

In React, use `onSelectedChange` and `onProvenanceChange`. Web Components use `selectedChange` and `provenanceChange`, with emitted values available in `event.detail`.
