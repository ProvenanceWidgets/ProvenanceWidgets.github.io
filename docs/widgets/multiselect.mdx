# Multiselect

A Multiselect is a user interface control element that allows users to select one or more options from a set of options.

```jsx
import { useState } from "react";
import {
  MultiSelectDropdown,
  ProvenanceButton,
} from "provenance-widgets";

const options = [
  { label: "New York", value: "New York" },
  { label: "Rome", value: "Rome" },
  { label: "London", value: "London" },
  { label: "Istanbul", value: "Istanbul" },
  { label: "Paris", value: "Paris" },
];

export default function MultiselectExample() {
  const [selected, setSelected] = useState(options.slice(0, 1));

  const provenanceChange = provenance => {
    console.log(provenance);
  };

  const selectedChange = nextSelected => {
    console.log(nextSelected);
    setSelected(nextSelected);
  };

  return (
    <div>
      <ProvenanceButton target="jsx-provenance-multiselect" />
      <MultiSelectDropdown
        id="jsx-provenance-multiselect"
        name="jsx-provenance-multiselect"
        options={options}
        selected={selected}
        optionLabel="label"
        dataKey="value"
        freeze={false}
        visualize={true}
        onProvenanceChange={provenanceChange}
        onSelectedChange={selectedChange}
      />
    </div>
  );
}
```

```js
// Create a new instance of the Multiselect
const multiselect = document.createElement(
  "web-provenance-multiselect",
);

const options = [
  { label: "New York", value: "New York" },
  { label: "Rome", value: "Rome" },
  { label: "London", value: "London" },
  { label: "Istanbul", value: "Istanbul" },
  { label: "Paris", value: "Paris" },
];

// Set the properties
multiselect.id = "webcomponent-provenance-multiselect";
multiselect.options = options;
multiselect.selected = options.slice(0, 1);
multiselect.optionLabel = "label";
multiselect.dataKey = "value";
multiselect.freeze = false;
multiselect.visualize = true;

multiselect.addEventListener("provenanceChange", event => {
  console.log(event.detail);
});

multiselect.addEventListener("selectedChange", event => {
  console.log(event.detail);
});

// Append it to the DOM
document.body.appendChild(multiselect);
```

### Events

In React, use `onSelectedChange` and `onProvenanceChange`. Web Components use `selectedChange` and `provenanceChange`, with emitted values available in `event.detail`.
