# Checkbox

A Checkbox is a user interface control element that allows users to select one or more options from a set of options.

```jsx
import { useState } from "react";
import { CheckboxGroup, ProvenanceButton } from "provenance-widgets";

const data = [
  { label: "New York", value: "New York" },
  { label: "Rome", value: "Rome" },
  { label: "London", value: "London" },
  { label: "Istanbul", value: "Istanbul" },
  { label: "Paris", value: "Paris" },
];

export default function CheckboxExample() {
  const [selected, setSelected] = useState(
    data.map(({ value }) => value).slice(0, 2),
  );

  const provenanceChange = provenance => {
    console.log(provenance);
  };

  const selectedChange = nextSelected => {
    console.log(nextSelected);
    setSelected(nextSelected);
  };

  return (
    <div>
      <ProvenanceButton target="jsx-provenance-checkbox" />
      <CheckboxGroup
        id="jsx-provenance-checkbox"
        name="jsx-provenance-checkbox"
        data={data}
        selected={selected}
        freeze={false}
        visualize={true}
        onProvenanceChange={provenanceChange}
        onSelectedChange={selectedChange}
      />
    </div>
  );
}
```

```js
// Create a new instance of the Checkbox
const checkbox = document.createElement("web-provenance-checkbox");

const data = [
  { label: "New York", value: "New York" },
  { label: "Rome", value: "Rome" },
  { label: "London", value: "London" },
  { label: "Istanbul", value: "Istanbul" },
  { label: "Paris", value: "Paris" },
];

// Set the properties
checkbox.id = "webcomponent-provenance-checkbox";
checkbox.name = "webcomponent-provenance-checkbox";
checkbox.data = data;
checkbox.selected = data.map(({ value }) => value).slice(0, 2);
checkbox.freeze = false;
checkbox.visualize = true;

checkbox.addEventListener("provenanceChange", event => {
  console.log(event.detail);
});

checkbox.addEventListener("selectedChange", event => {
  console.log(event.detail);
});

// Append it to the DOM
document.body.appendChild(checkbox);
```

### Events

In React, use `onSelectedChange` and `onProvenanceChange`. Web Components use `selectedChange` and `provenanceChange`, with emitted values available in `event.detail`.
